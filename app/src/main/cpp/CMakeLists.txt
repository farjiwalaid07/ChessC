cmake_minimum_required(VERSION 3.22.1)
project(chesscrazy)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SF_ZIP ${CMAKE_BINARY_DIR}/stockfish_flutter_plus.zip)
set(SF_DIR ${CMAKE_BINARY_DIR}/stockfish_flutter_plus-master)
if(NOT EXISTS ${SF_DIR}/ios/Stockfish/src/main.cpp)
  file(DOWNLOAD https://github.com/Dexterfury/stockfish_flutter_plus/archive/refs/heads/master.zip ${SF_ZIP} SHOW_PROGRESS STATUS SF_DOWNLOAD)
  list(GET SF_DOWNLOAD 0 SF_STATUS)
  if(NOT SF_STATUS EQUAL 0)
    message(FATAL_ERROR "Stockfish source download failed: ${SF_DOWNLOAD}")
  endif()
  execute_process(COMMAND ${CMAKE_COMMAND} -E tar xf ${SF_ZIP} WORKING_DIRECTORY ${CMAKE_BINARY_DIR} RESULT_VARIABLE SF_EXTRACT_RESULT)
  if(NOT SF_EXTRACT_RESULT EQUAL 0)
    message(FATAL_ERROR "Stockfish source extraction failed")
  endif()
endif()

file(GLOB_RECURSE STOCKFISH_CPP "${SF_DIR}/ios/Stockfish/src/*.cpp")
add_library(joyce_chess SHARED chess.cpp native.cpp stockfish_bridge.cpp ${SF_DIR}/ios/FlutterStockfish/ffi.cpp ${STOCKFISH_CPP})
target_include_directories(joyce_chess PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${SF_DIR}/ios/FlutterStockfish ${SF_DIR}/ios/Stockfish/src)
target_compile_definitions(joyce_chess PRIVATE USE_PTHREADS NDEBUG)
target_compile_options(joyce_chess PRIVATE -O3 -ffast-math)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64")
  target_link_options(joyce_chess PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
find_library(log-lib log)
target_link_libraries(joyce_chess ${log-lib})
